\posterblock{The Core Idea}{%
\normalsize
Natural images are \textbf{dense in pixels} but \textbf{sparse after a
wavelet / DCT transform}. We test the same idea for CNN activations: a
layer's dense $C\!\times\!H\!\times\!W$ feature map may be compressible
into a small set of learned \textbf{(feature, location)} coefficients.

\medskip
\textbf{Q-A:} Keep only the top-$m$ coefficients, reconstruct, splice
back into the frozen network. Does downstream accuracy hold?

\medskip
\textbf{Q-B:} Do the important sparse codes at layer $t$ \emph{causally
generate} the codes at layer $t{+}1$? Is there a depth-wise
\textbf{sparsity tree}?
}

\alertblock{Why It Matters}{%
\normalsize
\begin{itemize}
  \item CNN mechanistic interpretability lags far behind transformers.
  \item LM-SAEs (Anthropic, OpenAI) are now standard; CNN activations are
        spatial \emph{fields} — different inductive biases are needed.
  \item A sparsity tree would mean: understanding one layer's sparse code
        gives a causal handle on the \emph{entire network}.
\end{itemize}
}

\posterblock{SAE Designs}{%
\normalsize
\textbf{Type A — FieldSAE} \yes\par
$\mathbf{H} \to$ 1$\times$1 Conv $\to$ 3$\times$ LocalResBlock
(5$\times$5 DW conv, ConvNeXt-style) $\to$ TopK Mask $\to$
3$\times$ LocalResBlock $\to$ 1$\times$1 Conv $\to\hat{\mathbf{H}}$.
Learns a locally-mixed spatial transform before the sparse bottleneck.

\medskip
\textbf{Type B — VectorSAE}\par
Standard 1$\times$1 encoder/decoder applied independently at every
spatial location (shared weights). No spatial mixing.

\medskip
Both: $\mathbf{H}\to\mathbf{Z}\in\mathbb{R}^{K\times H\times W}
\to\hat{\mathbf{H}}$. \textbf{No skip connection} bypasses the sparse code.
}

\posterblock{Masking \& Training}{%
\normalsize
\textbf{Global TopK:} top-$m$ coefficients across the entire
$K{\times}H{\times}W$ field per image (budget = \% retained).

\smallskip
\textbf{RF-Local TopK:} top-$k_{\text{rf}}$ within each next-layer
receptive-field window — achieves 0.4--3.1\% effective retention.

\medskip
\textbf{Curriculum:}
\colorbox{LightGreen}{\footnotesize\bfseries Warmup 25\% (no mask)}
\colorbox{LightGold}{\footnotesize\bfseries Anneal 50\% (50\%$\to$target)}
\colorbox{LightOrange}{\footnotesize\bfseries Fixed 25\% (lock+refine)}

\smallskip
\textbf{Loss:} $\mathcal{L} = \mathrm{normMSE}(\hat{H},H) + 0.1(1-\cos(\hat{H},H))$

\medskip
\textbf{Baselines:} PCA at matched budget; Random-dictionary SAE.
}

\claimbox{2}{%
\textbf{$Q_{t+1}$'s important features are built from $Q_t$'s important features.}

Ablating top-magnitude parent coefficients damages downstream accuracy
\textbf{3--7.5$\times$ more} than random ablation (M1); child-code perturbations
concentrate \textbf{spatially within the parent's RF} at 3--57$\times$ above
chance (M2). This is a \textit{causal}, spatially-structured claim, not
merely a correlation.
}

\claimbox{3}{%
\textbf{Frozen SAE weights validate that the pipeline isolates IMPORTANT features.}

\textbf{Reasoning chain:}
\begin{enumerate}
  \item Frozen SAE weights $=$ the same transform on any input.
  \item Feed $\tilde{Q}_t$ (5\% sparse reconstruction) through frozen backbone
        + frozen SAE at $Q_{t+1}$.
  \item Task accuracy preserved to \textbf{71.2\%} ($-$0.6\,pp).
  \item Therefore: the same fixed transform, applied to the sparse support,
        identifies features $Q_{t+1}$ needs for task integrity.
  \item Since weights are frozen and accuracy holds, the sparse support must
        contain \textbf{truly important information} --- not just
        reconstruction-convenient feature vectors.
\end{enumerate}
This validates the core claim: \textbf{the pipeline isolates IMPORTANT features}.
}

\posterblock{Conclusions}{%
\normalsize
\begin{enumerate}
  \item \textbf{Sparse representation works} --- 2--5\% of learned
        coefficients suffice; beats PCA and random baselines decisively.
  \item \textbf{FieldSAE spatial mixing is essential} --- +5--8\,pp over
        VectorSAE at low budgets; only design surviving ultra-sparse RF masking.
  \item \textbf{Causal sparsity tree confirmed} --- 3--57$\times$ causal
        importance, 3--57$\times$ spatial concentration.
  \item \textbf{Re-grounding is the mechanism} --- BPTT reaches 70.1 vs
        71.2 causal bound.
  \item \textbf{Frozen-weight validation} --- same fixed transform
        consistently isolates load-bearing features.
\end{enumerate}

\textbf{Limitations:} Single backbone/dataset (R56/C100);
Q3$\to$Q4 bottleneck open; single training seed.
